\chapter*{Glosario y abreviaturas}
\addcontentsline{toc}{chapter}{Glosario y abreviaturas}

\begin{description}
  \item[EDSR] \emph{Enhanced Deep Residual Network}; red convolucional residual para
        superresolución.
  \item[HR] \emph{High Resolution}; imagen de alta resolución utilizada como referencia.
  \item[LR] \emph{Low Resolution}; imagen de baja resolución obtenida mediante degradación
        controlada.
  \item[OMR] \emph{Optical Music Recognition}; reconocimiento óptico de música.
  \item[PSNR] \emph{Peak Signal-to-Noise Ratio}; medida logarítmica de fidelidad basada en el
        error cuadrático medio.
  \item[SMB] \emph{Sheet Music Benchmark}; conjunto de partituras utilizado para la evaluación.
  \item[SSIM] \emph{Structural Similarity Index Measure}; medida de similitud estructural entre
        imágenes.
  \item[SR] \emph{Super-Resolution}; reconstrucción de una imagen a una resolución espacial
        superior.
\end{description}
